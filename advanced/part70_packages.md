## 1. .nimble File Format

```nimble
# mypackage.nimble
# Nimble package specification

# Package metadata
version     = "1.2.3"
author      = "Your Name"
description = "A library for doing awesome things"
license     = "MIT"

# Package configuration
srcDir      = "src"         # Source directory (default: "")
binDir      = "bin"         # Binary output directory
namedBin    = {"myapp": "src/myapp.nim"}  # Named executables
skipDirs    = @["examples", "benchmarks"]  # Dirs to skip when installing
skipFiles   = @["tests.nim"]               # Files to skip
skipExt     = @["md", "txt"]               # Extensions to skip

# Runtime dependencies
requires "nim >= 1.6.0"
requires "chronos >= 3.0.0 & < 4.0.0"  # Version range
requires "jsony"                         # Latest version

# Tasks
task test, "Run all tests":
  exec "nimble testSuite"

task testSuite, "Run unit tests":
  exec "nim c -r tests/test_all.nim"

task bench, "Run benchmarks":
  exec "nim c -d:release -r benchmarks/bench.nim"

task docs, "Generate documentation":
  exec "nim doc --project --index:on --outdir:docs/api src/mypackage.nim"

task clean, "Clean build artifacts":
  rmDir "bin"
  rmDir "nimcache"

# Custom hooks
before install:
  echo "Installing dependencies..."

after install:
  echo "Installation complete!"
```

## 3. API Design Principles

```nim
# src/mypackage/types.nim
# Design: clear types, proper errors, zero-cost where possible

type
  # Use distinct types to prevent accidental misuse
  UserId* = distinct int
  Email* = distinct string
  
  # Errors should be specific
  MyError* = object of CatchableError
  ValidationError* = object of MyError
    field*: string
  NotFoundError* = object of MyError
    id*: string
  
  # Use Option for optional values (not nil)
  Config* = object
    host*: string
    port*: int
    timeout*: int
    maxRetries*: int
    debug*: bool
  
  Result*[T, E] = object
    case isOk*: bool
    of true:  value*: T
    of false: error*: E

# Constructors hide implementation details
proc newConfig*(host: string, port = 8080): Config =
  Config(host: host, port: port, timeout: 30, maxRetries: 3)

proc ok*[T, E](value: T): Result[T, E] =
  Result[T, E](isOk: true, value: value)

proc err*[T, E](error: E): Result[T, E] =
  Result[T, E](isOk: false, error: error)

proc get*[T, E](r: Result[T, E]): T =
  if not r.isOk:
    raise newException(ValueError, "Result is Err")
  r.value

proc getOrDefault*[T, E](r: Result[T, E], default: T): T =
  if r.isOk: r.value else: default

# Use concepts for flexible APIs
type
  Serializable* = concept x
    $x is string

  Comparable*[T] = concept x
    x < x is bool
    x == x is bool

proc sortAndStringify*[T: Serializable and Comparable[T]](items: seq[T]): seq[string] =
  var sorted = items
  sorted.sort()
  sorted.mapIt($it)
```

## 4. Documentation Generation

```nim
# src/mypackage/core.nim
## Core functionality of MyPackage.
##
## This module provides the main operations:
## - `doSomething`: Process input and return result
## - `batchProcess`: Process multiple inputs efficiently
##
## Thread Safety:
## All procedures in this module are **thread-safe** unless noted otherwise.

import std/[options, strformat, strutils]
import ./types

proc doSomething*(input: string): Result[string, string] =
  ## Process `input` and return transformed result.
  ##
  ## The transformation:
  ## 1. Validates input is not empty
  ## 2. Converts to uppercase
  ## 3. Trims whitespace
  ##
  ## **Parameters:**
  ## - `input`: The string to process
  ##
  ## **Returns:**
  ## `Result[string, string]` — Ok with transformed value, or Err with error message
  ##
  ## **Example:**
  ## ```nim
  ## let r = doSomething("  hello  ")
  ## assert r.isOk
  ## assert r.get() == "HELLO"
  ## ```
  ##
  ## **See also:**
  ## - `batchProcess`
  
  if input.len == 0:
    return err[string, string]("Input cannot be empty")
  
  ok[string, string](input.strip().toUpperAscii())

proc batchProcess*(inputs: seq[string],
    failFast = false): tuple[results: seq[Result[string, string]], errors: int] =
  ## Process multiple inputs, optionally stopping on first error.
  ##
  ## **Parameters:**
  ## - `inputs`: Sequence of strings to process
  ## - `failFast`: If true, stop processing after first error
  ##
  ## **Returns:**
  ## Tuple with results seq and count of errors
  var results: seq[Result[string, string]]
  var errorCount = 0
  
  for input in inputs:
    let r = doSomething(input)
    results.add(r)
    if not r.isOk:
      inc errorCount
      if failFast: break
  
  (results: results, errors: errorCount)

# Internal implementation (not exported)
proc internalHelper(s: string): string =
  s.strip()

# Generate docs:
# nim doc --project --index:on --outdir:docs/api src/mypackage.nim
```

## 5. Testing Infrastructure

```nim
# tests/test_core.nim
import std/[unittest, options]
import mypackage/core
import mypackage/types

suite "doSomething":
  test "processes valid input":
    let r = doSomething("hello")
    check r.isOk
    check r.get() == "HELLO"
  
  test "trims whitespace":
    let r = doSomething("  hello  ")
    check r.isOk
    check r.get() == "HELLO"
  
  test "returns error for empty input":
    let r = doSomething("")
    check not r.isOk
    check r.error == "Input cannot be empty"
  
  test "handles unicode":
    let r = doSomething("café")
    check r.isOk

suite "batchProcess":
  test "processes all valid inputs":
    let (results, errors) = batchProcess(@["a", "b", "c"])
    check errors == 0
    check results.len == 3
  
  test "counts errors":
    let (results, errors) = batchProcess(@["a", "", "c"])
    check errors == 1
    check results.len == 3
  
  test "failFast stops on first error":
    let (results, errors) = batchProcess(@["a", "", "c"], failFast = true)
    check errors == 1
    check results.len == 2  # Stopped after second element

# tests/config.nims — Test configuration
# switch("path", parentDir(thisDir()) / "src")
# switch("define", "testMode")
```

## 6. Benchmarking

```nim
# benchmarks/bench.nim
import std/[times, strformat, math]
import mypackage/core

type
  BenchResult* = object
    name*: string
    iterations*: int
    totalNs*: float
    avgNs*: float
    minNs*: float
    maxNs*: float
    opsPerSec*: float

proc benchmark*(name: string, iterations: int,
    fn: proc()): BenchResult =
  var times: seq[float]
  
  # Warmup
  for _ in 0..min(iterations div 10, 100):
    fn()
  
  for _ in 0..<iterations:
    let t0 = cpuTime()
    fn()
    let t1 = cpuTime()
    times.add((t1 - t0) * 1e9)  # Convert to nanoseconds
  
  let total = times.foldl(a + b, 0.0)
  let avg = total / float(iterations)
  let minT = times.foldl(min(a, b), times[0])
  let maxT = times.foldl(max(a, b), times[0])
  
  BenchResult(
    name: name,
    iterations: iterations,
    totalNs: total,
    avgNs: avg,
    minNs: minT,
    maxNs: maxT,
    opsPerSec: 1e9 / avg
  )

proc printResult*(r: BenchResult) =
  echo &"{r.name}:"
  echo &"  Iterations: {r.iterations}"
  echo &"  Avg:        {r.avgNs:.1f} ns"
  echo &"  Min:        {r.minNs:.1f} ns"
  echo &"  Max:        {r.maxNs:.1f} ns"
  echo &"  Ops/sec:    {r.opsPerSec:.0f}"

when isMainModule:
  let r1 = benchmark("doSomething (short)", 100_000, proc() =
    discard doSomething("hello world")
  )
  printResult(r1)
  
  let longStr = "x".repeat(1000)
  let r2 = benchmark("doSomething (1000 chars)", 10_000, proc() =
    discard doSomething(longStr)
  )
  printResult(r2)
```

## 7. Version Management & Changelog

```nim
# src/mypackage/version.nim
## Version information for MyPackage

const
  Version* = "1.2.3"
  VersionMajor* = 1
  VersionMinor* = 2
  VersionPatch* = 3
  
  # Semantic versioning rules:
  # Major: breaking API changes
  # Minor: new features (backward compatible)
  # Patch: bug fixes (backward compatible)

proc checkCompatibility*(requiredVersion: string): bool =
  ## Check if installed version satisfies requirement
  let parts = requiredVersion.split('.')
  if parts.len != 3: return false
  
  let reqMajor = parseInt(parts[0])
  let reqMinor = parseInt(parts[1])
  
  if reqMajor != VersionMajor: return false  # Major must match
  if VersionMinor < reqMinor: return false   # Minor must be >= required
  true
```

## 8. CI/CD Configuration

```yaml
# .github/workflows/ci.yml
name: CI

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]

jobs:
  test:
    name: Test on ${{ matrix.os }} with Nim ${{ matrix.nim }}
    runs-on: ${{ matrix.os }}
    
    strategy:
      matrix:
        os: [ubuntu-latest, macos-latest, windows-latest]
        nim: ["1.6.14", "2.0.0", "stable"]
    
    steps:
    - uses: actions/checkout@v3
    
    - name: Setup Nim
      uses: jiro4989/setup-nim-action@v1
      with:
        nim-version: ${{ matrix.nim }}
        repo-token: ${{ secrets.GITHUB_TOKEN }}
    
    - name: Install dependencies
      run: nimble install -y
    
    - name: Run tests
      run: nimble test
    
    - name: Run benchmarks (release mode)
      run: nim c -d:release -r benchmarks/bench.nim
    
    - name: Check documentation compiles
      run: nim doc --project --index:on --outdir:/tmp/docs src/mypackage.nim

  publish:
    name: Publish to packages.nim.org
    needs: test
    runs-on: ubuntu-latest
    if: startsWith(github.ref, 'refs/tags/v')
    
    steps:
    - uses: actions/checkout@v3
    - name: Setup Nim
      uses: jiro4989/setup-nim-action@v1
    - name: Publish
      run: nimble publish
      env:
        NIMBLE_PUBLISH_TOKEN: ${{ secrets.NIMBLE_PUBLISH_TOKEN }}
```

## 9. config.nims — Project-wide Nim settings

```nim
# config.nims
# Project-wide Nim compiler settings (checked into git)

# Always enable all warnings
switch("warning", "all:on")

# Treat warnings as errors in CI
when defined(ci):
  switch("warning", "error:on")

# Use ORC GC (best for most cases)
switch("mm", "orc")

# Strict case
switch("styleCheck", "error")

# Enable multi-threading
switch("threads", "on")

# Custom defines
when defined(testMode):
  echo "Building in test mode"
  switch("define", "debugLogging")

# Path aliases
switch("path", thisDir() / "src")

# Release mode optimizations
when defined(release):
  switch("opt", "speed")
  switch("define", "ssl")  # Enable SSL in release

# Platform-specific
when defined(windows):
  switch("define", "useWinCrt")
elif defined(macosx):
  switch("define", "useCoreFoundation")
```
